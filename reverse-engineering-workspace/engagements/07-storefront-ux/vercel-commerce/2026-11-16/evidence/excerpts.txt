EVIDENCE 5 - variant-selector.tsx: URL-search-params state, availability-aware
Source: components/product/variant-selector.tsx (full, 106 lines)
Proves: the variant selector:
  1) Uses URL search params (?size=M&color=blue) as state, NOT React state.
     This makes the selected variant SHAREABLE (copy the URL).
  2) Computes a 'combinations' map from all variants upfront. For each option
     value, checks if any matching variant is availableForSale.
  3) Unavailable options get aria-disabled + disabled + a strike-through CSS
     pseudo-element (the before:-rotate-45 line). Title shows 'Out of Stock'.
  4) Returns null if there are no options or only one option with one value
     (simple products don't show a selector).
  Rule: variant state belongs in the URL, not in React state. Unavailable
  variants are visible-but-disabled (not hidden) so the user sees what's OOS.
-------------------------------------------------------------------------
     1	"use client";
     2	
     3	import clsx from "clsx";
     4	import { ProductOption, ProductVariant } from "lib/shopify/types";
     5	import { useRouter, useSearchParams } from "next/navigation";
     6	
     7	type Combination = {
     8	  id: string;
     9	  availableForSale: boolean;
    10	  [key: string]: string | boolean;
    11	};
    12	
    13	export function VariantSelector({
    14	  options,
    15	  variants,
    16	}: {
    17	  options: ProductOption[];
    18	  variants: ProductVariant[];
    19	}) {
    20	  const router = useRouter();
    21	  const searchParams = useSearchParams();
    22	  const hasNoOptionsOrJustOneOption =
    23	    !options.length ||
    24	    (options.length === 1 && options[0]?.values.length === 1);
    25	
    26	  if (hasNoOptionsOrJustOneOption) {
    27	    return null;
    28	  }
    29	
    30	  const combinations: Combination[] = variants.map((variant) => ({
    31	    id: variant.id,
    32	    availableForSale: variant.availableForSale,
    33	    ...variant.selectedOptions.reduce(
    34	      (accumulator, option) => ({
    35	        ...accumulator,
    36	        [option.name.toLowerCase()]: option.value,
    37	      }),
    38	      {},
    39	    ),
    40	  }));
    41	
    42	  const updateOption = (name: string, value: string) => {
    43	    const params = new URLSearchParams(searchParams.toString());
    44	    params.set(name, value);
    45	    router.replace(`?${params.toString()}`, { scroll: false });
    46	  };
    47	
    48	  return options.map((option) => (
    49	    <form key={option.id}>
    50	      <dl className="mb-8">
    51	        <dt className="mb-4 text-sm uppercase tracking-wide">{option.name}</dt>
    52	        <dd className="flex flex-wrap gap-3">
    53	          {option.values.map((value) => {
    54	            const optionNameLowerCase = option.name.toLowerCase();
    55	
    56	            // Base option params on current searchParams so we can preserve any other param state.
    57	            const optionParams: Record<string, string> = {};
    58	            searchParams.forEach((v, k) => (optionParams[k] = v));
    59	            optionParams[optionNameLowerCase] = value;
    60	
    61	            // Filter out invalid options and check if the option combination is available for sale.
    62	            const filtered = Object.entries(optionParams).filter(
    63	              ([key, value]) =>
    64	                options.find(
    65	                  (option) =>
    66	                    option.name.toLowerCase() === key &&
    67	                    option.values.includes(value),
    68	                ),
    69	            );
    70	            const isAvailableForSale = combinations.find((combination) =>
    71	              filtered.every(
    72	                ([key, value]) =>
    73	                  combination[key] === value && combination.availableForSale,
    74	              ),
    75	            );
    76	
    77	            // The option is active if it's in the selected options.
    78	            const isActive = searchParams.get(optionNameLowerCase) === value;
    79	
    80	            return (
    81	              <button
    82	                formAction={() => updateOption(optionNameLowerCase, value)}
    83	                key={value}
    84	                aria-disabled={!isAvailableForSale}
    85	                disabled={!isAvailableForSale}
    86	                title={`${option.name} ${value}${!isAvailableForSale ? " (Out of Stock)" : ""}`}
    87	                className={clsx(
    88	                  "flex min-w-[48px] items-center justify-center rounded-full border bg-neutral-100 px-2 py-1 text-sm dark:border-neutral-800 dark:bg-neutral-900",
    89	                  {
    90	                    "cursor-default ring-2 ring-blue-600": isActive,
    91	                    "ring-1 ring-transparent transition duration-300 ease-in-out hover:ring-blue-600":
    92	                      !isActive && isAvailableForSale,
    93	                    "relative z-10 cursor-not-allowed overflow-hidden bg-neutral-100 text-neutral-500 ring-1 ring-neutral-300 before:absolute before:inset-x-0 before:-z-10 before:h-px before:-rotate-45 before:bg-neutral-300 before:transition-transform dark:bg-neutral-900 dark:text-neutral-400 dark:ring-neutral-700 dark:before:bg-neutral-700":
    94	                      !isAvailableForSale,
    95	                  },
    96	                )}
    97	              >
    98	                {value}
    99	              </button>
   100	            );
   101	          })}
   102	        </dd>
   103	      </dl>
   104	    </form>
   105	  ));
   106	}

EVIDENCE 6 - Shopify product type (the data shape the UI consumes)
Source: lib/shopify/types.ts + lib/shopify/fragments/product.ts
Proves: the Product type has:
  - id, handle, title, description, descriptionHtml
  - availableForSale (bool), totalInventory
  - priceRange { minVariantPrice, maxVariantPrice } (each {amount, currencyCode})
  - featuredImage { url, width, height, altText }
  - images[] (gallery)
  - options[] (variant dimensions like Color, Size)
  - variants[] (each with selectedOptions + availableForSale + price)

  model - we have ASIN, title, price, image, marketplace. The 'variants' and
  'availableForSale' fields are more relevant to a real cart; for an affiliate
  funnel we only need: title, price, image, ASIN, destination URL, availability.
