EVIDENCE 4 — door-key-validate.ts: the sliding session (Lock #2)
Source: src/lib/auth/door-key-validate.ts (full, 69 lines)
Proves: validateDoorKey() reads the cookie, SHA-256 hashes it, queries
        admin_access_keys WHERE token_hash = ? AND kind = 'door' AND
        revoked_at IS NULL AND expires_at > now(). If found, slides the
        expiry forward 30 days (sliding renewal). If not found → ok: false.
        Rule: a door key is a long-lived bearer token whose hash is in the DB.
        Sliding renewal = active admins never have to re-issue; idle admins
        expire after 30 days of no use.
-------------------------------------------------------------------------
     1	import { createHash } from 'node:crypto';
     2	import { createAdminClient } from '@/lib/supabase/server';
     3	import { cookies } from 'next/headers';
     4	
     5	import { DOOR_KEY_COOKIE_NAME } from '@/lib/auth/door-key';
     6	
     7	/**
     8	 * Validate the door-key cookie on the current request.
     9	 *
    10	 * Returns { ok: true } when a live, unused, unrevoked door key matches the
    11	 * cookie, and slides its expiry forward by 30 days (sliding renewal).
    12	 *
    13	 * Returns { ok: false } when the cookie is missing, wrong, used, revoked, or
    14	 * expired — the caller must return 404.
    15	 */
    16	export async function validateDoorKey(): Promise<{ ok: boolean }> {
    17	  const cookieStore = await cookies();
    18	  const token = cookieStore.get(DOOR_KEY_COOKIE_NAME)?.value;
    19	
    20	  if (!token) {
    21	    return { ok: false };
    22	  }
    23	
    24	  const tokenHash = createHash('sha256').update(token).digest('hex');
    25	
    26	  // admin_access_keys isn't in the generated DB types yet — cast through any.
    27	  const supabase = createAdminClient() as unknown as {
    28	    from: (table: string) => AnyChain;
    29	  };
    30	
    31	  type AnyChain = {
    32	    select: (cols: string) => AnyChain;
    33	    eq: (col: string, val: string | number | null) => AnyChain;
    34	    is: (col: string, val: null) => AnyChain;
    35	    gt: (col: string, val: string) => AnyChain;
    36	    maybeSingle: () => Promise<{ data: any | null; error: unknown }>;
    37	    update: (vals: Record<string, unknown>) => AnyUpdateChain;
    38	  };
    39	
    40	  type AnyUpdateChain = {
    41	    eq: (col: string, val: string | number) => Promise<{ error: unknown }>;
    42	  };
    43	
    44	  const { data: key, error: findError } = await supabase
    45	    .from('admin_access_keys')
    46	    .select('id, user_id, expires_at, used_at, revoked_at, kind')
    47	    .eq('token_hash', tokenHash)
    48	    .eq('kind', 'door')
    49	    .is('revoked_at', null)
    50	    .gt('expires_at', new Date().toISOString())
    51	    .maybeSingle();
    52	
    53	  if (findError || !key) {
    54	    return { ok: false };
    55	  }
    56	
    57	  // Sliding renewal: push expiry to now() + 30 days on each valid visit.
    58	  const newExpiry = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();
    59	  const { error: updateError } = await supabase
    60	    .from('admin_access_keys')
    61	    .update({ expires_at: newExpiry })
    62	    .eq('id', key.id);
    63	
    64	  if (updateError) {
    65	    return { ok: false };
    66	  }
    67	
    68	  return { ok: true };
    69	}

EVIDENCE 6 — signup route: invite-code redemption + door-key issuance
Source: src/app/api/admin/auth/signup/route.ts (full, 110 lines)
Proves: Lock #1 (invite code). The signup flow:
        1) Hash the submitted invite code, look up admin_access_keys WHERE
           kind = 'invite' AND used_at IS NULL AND revoked_at IS NULL AND
           expires_at > now(). Re-validate at submit time (prevents races).
        2) Create the user via supabase.auth.admin.createUser (service role).
        3) Set app_metadata.role = 'admin' via updateUserById (service role).
        4) Mark the invite code as used_at = now().
        5) Issue a NEW door key: hash a fresh random token, INSERT into
           admin_access_keys (kind='door', user_id, token_hash, expires_at).
        6) Set the door-key cookie, redirect to /admin/signin.
        Rule: invite codes are one-time, hashed, revocable, expiring.
