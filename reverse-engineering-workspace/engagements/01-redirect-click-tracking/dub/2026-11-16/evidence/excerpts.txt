    13	export const config = {
    14	  runtime: "nodejs",
    15	  matcher: [
    16	    /*
    17	     * Match all paths except for:
    18	     * 1. /api/ routes
    19	     * 2. /_next/ (Next.js internals)
    20	     * 3. /_proxy/ (proxies for third-party services)
    21	     * 4. Metadata files: favicon.ico, sitemap.xml, robots.txt, manifest.webmanifest

    28	  const { domain, path, key, fullKey } = parse(req);
    29	
    30	  // Axiom logging
    31	  logger.info(...transformMiddlewareRequest(req));
    32	  ev.waitUntil(logger.flush());
    33	
    34	  // for app.dub.co
    35	  if (isAppHostname(domain)) {
    36	    return AppMiddleware(req);
    37	  }
    38	
    39	  // for api.dub.co
    40	  if (API_HOSTNAMES.has(domain)) {
    41	    return ApiMiddleware(req);
    42	  }
    43	
    44	  // for public stats pages (e.g. d.to/stats/try -> rewrite to [/domain]/[key]/stats)
    45	  if (path.startsWith("/stats/")) {
    46	    return NextResponse.rewrite(
    47	      new URL(
    48	        `/${domain}/${encodeURIComponent(path.replace("/stats/", ""))}/stats`,
    49	        req.url,
    50	      ),
    51	    );
    52	  }
    53	
    54	  // for .well-known routes
    55	  if (path.startsWith("/.well-known/")) {
    56	    const file = path.split("/.well-known/").pop();
    57	    if (file && supportedWellKnownFiles.includes(file)) {
    58	      return NextResponse.rewrite(
    59	        new URL(`/wellknown/${domain}/${file}`, req.url),
    60	      );
    61	    }
    62	  }
    63	
    64	  // default redirects for dub.sh
    65	  if (domain === "dub.sh" && DEFAULT_REDIRECTS[key]) {
    66	    return NextResponse.redirect(DEFAULT_REDIRECTS[key]);
    67	  }
    68	
    69	  if (isValidUrl(fullKey)) {
    70	    return CreateLinkMiddleware(req);
    71	  }
    72	
    73	  return LinkMiddleware(req, ev);
    74	}

     1	import { ipAddress } from "@vercel/functions";
     2	import { userAgent } from "next/server";
     3	import { IP_BOTS, IP_RANGES_BOTS, REFERRER_BOTS, UA_BOTS } from "./bots-list";
     4	import { isIpInRange } from "./is-ip-in-range";
     5	
     6	// UA fragments that look like a bot pattern but aren't – e.g. Instagram's
     7	// in-app webview on Pixel devices appends "Google/google" in its device
     8	// descriptor, which would otherwise trip the generic "google" entry in UA_BOTS
     9	const UA_FALSE_POSITIVES = [/Google\/google\b/];
    10	
    11	export const detectBot = (req: Request) => {
    12	  const searchParams = new URL(req.url).searchParams;
    13	
    14	  if (searchParams.get("bot")) {
    15	    return true;
    16	  }
    17	
    18	  // HEAD requests are generally from bots, real users will always use GET requests
    19	  if (req.method === "HEAD") {
    20	    return true;
    21	  }
    22	
    23	  // Check ua
    24	  const ua = userAgent(req);
    25	
    26	  if (ua) {
    27	    const isKnownFalsePositive = UA_FALSE_POSITIVES.some((pattern) =>
    28	      pattern.test(ua.ua),
    29	    );
    30	    const sanitizedUa = UA_FALSE_POSITIVES.reduce(
    31	      (s, pattern) => s.replace(pattern, ""),
    32	      ua.ua,
    33	    );
    34	    return (
    35	      (!isKnownFalsePositive && ua.isBot) ||
    36	      UA_BOTS.some((bot) => new RegExp(bot, "i").test(sanitizedUa))
    37	    );
    38	  }
    39	
    40	  // Check referer
    41	  const referer = req.headers.get("referer");
    42	  if (
    43	    referer &&
    44	    REFERRER_BOTS.some((bot) => new RegExp(bot, "i").test(referer))
    45	  ) {
    46	    return true;
    47	  }
    48	
    49	  // Check ip
    50	  let ip = ipAddress(req);
    51	
    52	  if (!ip) {
    53	    return false;
    54	  }
    55	
    56	  // Check exact IP matches
    57	  if (IP_BOTS.includes(ip)) {
    58	    return true;
    59	  }
    60	
    61	  // Check CIDR ranges
    62	  return IP_RANGES_BOTS.some((range) => isIpInRange(ip, range));
    63	};
