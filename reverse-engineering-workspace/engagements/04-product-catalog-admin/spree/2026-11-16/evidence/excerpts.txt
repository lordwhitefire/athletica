EVIDENCE 2 — available? gate (the publish rule)
Source: samples/spree/spree/core/app/models/spree/product.rb:700-728
Proves: a product is available? when:
        1) status == 'active' (NOT 'draft', NOT 'archived', NOT 'proposed', NOT 'rejected')
        2) NOT deleted (soft-delete via acts_as_paranoid)
        3) available_on is nil OR available_on <= now (future-publish scheduling)
        Rule: THREE conditions must hold for a product to be live in the catalog.
-------------------------------------------------------------------------
   700	    def available?
   701	      active? && !deleted? && (available_on.nil? || available_on <= Time.current)
   702	    end
   703	
   704	    # True when the product is currently offered as a pre-order: at least one
   705	    # variant is sold before it is in stock (see {Variant#preorder?}).
   706	    # Independent of publishing — a pre-order product is live in the catalog,
   707	    # just flagged "buy now, ships later".
   708	    # @return [Boolean]
   709	    def preorder?
   710	      default_variant.preorder? || variants.any?(&:preorder?)
   711	    end
   712	
   713	    # The latest "ships by" date across the product's pre-order variants, so a
   714	    # storefront can show a single "ships by" promise for the product. Nil when
   715	    # the product is not a pre-order. Scans the same sellable set {#preorder?}
   716	    # consults (the master only counts when it is the sole, purchasable
   717	    # variant), reading preloaded associations to avoid an N+1.

EVIDENCE 10 — BulkOperations concern (the bulk pattern)
Source: samples/spree/spree/api/app/controllers/concerns/spree/api/v3/bulk_operations.rb
Proves: shared concern included by products_controller. Provides:
        - require_ids! (validates ids array present)
        - bulk_add_tags / bulk_remove_tags
        - bulk_collection (the scoped ActiveRecord relation)
        - bulk_tags_response (the JSON shape)
        - decode_ids (handles encoded id formats)
        Rule: bulk actions live in a shared concern, not duplicated per resource.
-------------------------------------------------------------------------
     1	module Spree
     2	  module Api
     3	    module V3
     4	      # Provides the shared tag bulk actions, the `bulk_collection` relation,
     5	      # and the `after_bulk_tags_change` hook controllers override (e.g. to
     6	      # reindex search + match automatic taxons).
     7	      module BulkOperations
     8	        extend ActiveSupport::Concern
     9	
    10	        included do
    11	          # Guards the concern's two actions; host controllers add their own
    12	          # bulk actions to the filter with `before_action :require_ids!, only: [...]`.
    13	          before_action :require_ids!, only: [:bulk_add_tags, :bulk_remove_tags]
    14	        end
    15	
    16	        # POST /api/v3/admin/<resource>/bulk_add_tags
    17	        # Body: { ids: [...], tags: ['summer', 'sale'] }
    18	        def bulk_add_tags
    19	          authorize! :update, model_class
    20	
    21	          Spree::Tags::BulkAdd.call(tag_names: Array(params[:tags]), records: bulk_collection)
    22	          after_bulk_tags_change
    23	
    24	          render json: bulk_tags_response
    25	        end
    26	
    27	        # POST /api/v3/admin/<resource>/bulk_remove_tags
    28	        # Body: { ids: [...], tags: ['summer', 'sale'] }
    29	        def bulk_remove_tags
    30	          authorize! :update, model_class
    31	
    32	          Spree::Tags::BulkRemove.call(tag_names: Array(params[:tags]), records: bulk_collection)
    33	          after_bulk_tags_change
    34	
    35	          render json: bulk_tags_response
    36	        end
    37	
    38	        private
    39	
    40	        # 422s when the caller omits `ids` entirely. An explicit empty array
    41	        # is allowed (= no-op, useful when a UI clears its selection between
    42	        # render and submit); a missing key is treated as a client bug since
    43	        # bulk endpoints would otherwise silently match no rows and the
    44	        # caller couldn't tell the difference from a successful zero-match.
    45	        def require_ids!
    46	          return if params.key?(:ids)
    47	
    48	          render_error(
    49	            code: 'missing_ids',
    50	            message: 'ids is required (send an empty array to no-op).',
    51	            status: :unprocessable_content
    52	          )
    53	        end
    54	
    55	        # Hook for controllers to perform additional work after bulk tag
    56	        # mutations — e.g. enqueueing search reindex jobs, re-matching
    57	        # automatic taxons.
    58	        def after_bulk_tags_change
    59	        end
    60	
    61	        # Slim ability-scoped relation targeted by the inbound `ids` param.
    62	        # Cross-store IDs and IDs the current admin can't update are silently
    63	        # dropped before any mutation runs. Mirrors the legacy concern's
    64	        # `bulk_collection` — no eager loads, because bulk endpoints only need
    65	        # `id` + `store_ids` per record.
    66	        def bulk_collection
    67	          @bulk_collection ||= begin
    68	            product_ids = decode_ids(params[:ids])
    69	            model_class.for_store(current_store).accessible_by(current_ability, :update)
    70	              .where(id: product_ids)
    71	          end
    72	        end
    73	
    74	        # Default shape for bulk tag responses; controllers override
    75	        # `bulk_record_count_key` to give the count its resource-named JSON key
    76	        # (e.g. `product_count`, `order_count`). Mirrors how the legacy admin
    77	        # concern delegates response shaping to the host controller.
    78	        def bulk_tags_response
    79	          {
    80	            bulk_record_count_key => bulk_collection.size,
    81	            tag_count: Array(params[:tags]).size
    82	          }
    83	        end
    84	
    85	        def bulk_record_count_key
    86	          :record_count
    87	        end
    88	
    89	        # Maps inbound IDs (a mix of prefixed `prod_…` strings and raw IDs) to
    90	        # raw IDs. Anything that isn't a valid prefixed ID is passed through
    91	        # verbatim, so legacy clients sending raw IDs keep working. Prefix
    92	        # decoding is pure — no DB lookup needed.
    93	        def decode_ids(ids)
    94	          Array(ids).map do |id|
    95	            Spree::PrefixedId.prefixed_id?(id) ? Spree::PrefixedId.decode_prefixed_id(id) : id
    96	          end
    97	        end
    98	      end
